# Every knob bin/spawn.sh's own header documents for a project file, in the order a
# forker meets them there. Copy this to projects/<name>.env (dropping the .txt — the
# installer's and spawner's own globs are both `*.env`, so this file, left named `.txt`,
# is never read by anything live; see projects/README.md) and fill in the six lines
# marked "required"; everything else is optional and shown at its own default.
#
# A project with one lane needs nothing past BOARD_NUMBER — see riffmap.env for the
# shortest real example. deploy/add-project.sh writes exactly that shortest form for you;
# reach for this file when you want a second lane (projects/example-triage.env.txt), the
# canary channel, or to see what every knob actually does before changing one.

# required. State and log directories are keyed on this.
PROJECT_NAME=example

# required. This project's own checkout on the box — a SEPARATE clone from /opt/watcher,
# never that path itself. Card worktrees are made beside it.
PROJECT_ROOT=/home/watcher/projects/example

# optional. owner/repo the cards live in — the installer clones this, and every
# repo-scoped `gh` call (claims, comments, the dependency reader) is pinned to it.
# Defaults to this checkout's own git remote "origin", so this is only needed when that
# remote is not the right repo (or there isn't one yet).
PROJECT_REPO=example-owner/example-repo

# required. GitHub user/org that owns the Projects board (board.py reads this with
# BOARD_NUMBER).
BOARD_OWNER=example-owner

# required. The board's number, from its own URL
# (https://github.com/users/<owner>/projects/<NUMBER>, or orgs/<owner>/projects/<NUMBER>).
BOARD_NUMBER=1

# optional. The GitHub login every Stuck/park/awaiting-rebuild comment addresses
# ("Needs @<login>:"). Defaults to BOARD_OWNER when unset, so a fork pings its own owner
# with no config at all — set this only when the board's owner and the person who should
# be asked are different accounts.
OWNER_LOGIN=example-owner

# optional. A per-project Discord channel for stuck/broken alerts (bin/notify.py). The
# box's systemd unit loads /etc/watcher/env (the shared, box-wide value) BEFORE sourcing
# this file, so leave the line commented out to inherit that value — an empty assignment
# here would silently clear it for this project alone. Uncomment only to point this one
# project at a different channel, and never commit a real webhook URL.
#DISCORD_WEBHOOK_URL=

# optional. Who a Stuck alert pings: "<@id>" (a user), "<@&roleid>" (a role), or "@here".
# Same inheritance rule as DISCORD_WEBHOOK_URL above — leave commented to inherit the
# box-wide value, uncomment only to override it for this project.
#DISCORD_MENTION=

# optional. The same stuck/broken alerts, to a Slack channel's incoming webhook instead
# of or alongside Discord (bin/notify.py; both set posts to both). Same inheritance rule
# as DISCORD_WEBHOOK_URL above.
#SLACK_WEBHOOK_URL=

# optional. Who a Stuck alert pings on Slack: "<@U…>" (a user) or "<!here>". Same
# inheritance rule as DISCORD_MENTION above.
#SLACK_MENTION=

# optional. Which cards THIS instance may pick, by `agent:` label — a comma-separated
# list ("architect,designer"), the same list prefixed with "!" for every card EXCEPT
# those ("!architect,designer"), or "-" alone for a card with no agent label at all.
# Unset means every card: the right default for a project with a single lane. See
# projects/example-triage.env.txt for a second lane that only takes "-".
LANE_AGENTS=

# optional. The slash command this lane's tick runs. Default: /watch. A triage lane sets
# /triage (projects/example-triage.env.txt).
LANE_COMMAND=/watch

# optional. "card" (the default) runs the tick in a worktree made for the card; "root"
# runs it in PROJECT_ROOT with no worktree at all — for a lane whose tick only touches
# the board, never the code (a triage lane; see commands/triage.md).
LANE_WORKDIR=card

# optional. This project's own gitignored, in-tree private inputs — space-separated,
# relative to the checkout — copied into every card worktree so a tick can read them
# (bin/worktree.sh). Default: none. riffmap's own is its real meeting transcripts
# directory; a project with nothing like that leaves this empty.
WORKTREE_COPY_DIRS=

# optional. 1 = also pull from the Ready column when nothing is claimable in In progress
# or Waiting, so a card moves on its own instead of waiting for a human to drag it.
# Default: 0 (off).
READY_PULL=0

# optional. With READY_PULL=1, the most cards this lane lets sit In progress before it
# stops promoting from Ready — counts the whole board's In progress column, not just this
# lane's own cards. Default: 2.
WIP_MAX=2

# optional. --max-turns for the tick. Default: 400.
TICK_MAX_TURNS=400

# optional. --max-budget-usd for the tick. On the box's Claude subscription this is a
# QUOTA ceiling, never a bill — it only stops a tick that is looping. Default: 15.
TICK_MAX_BUDGET_USD=15

# optional. The watchdog ceiling, in minutes, that kills a tick stuck with no progress.
# Default: 150.
TICK_TIMEOUT=150

# optional. The daemon's poll cadence, in seconds — how often an idle lane checks the
# board for a claimable card. Default: 60.
POLL_INTERVAL=60

# optional. "canary" runs this lane from the box's second, independent checkout
# (/opt/watcher-canary) instead of the one every other lane shares, so a bad change
# reaches this lane's board before it reaches every other project. Default: unset
# ("stable") — the checkout spawn.sh itself is already running from. Leave this alone
# unless you are deliberately opting a lane into the canary stage
# (deploy/autodeploy.sh, "the canary stage").
LANE_CHANNEL=stable
